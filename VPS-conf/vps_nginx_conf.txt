# * the command to get ssl certs for websites that will also auto renew is:
# ! sudo certbot certonly --webroot -w /var/www/certbot -d {domain url} -d www.{domain url}

map $http_upgrade $connection_upgrade {
    default upgrade;
    ''      close;
}

upstream bgmi-frontend {
    server hypedin-BGMI_frontend:3000;
}

upstream bgmi-backend {
    server hypedin-BGMI_backend:3000;
}

upstream level-up_frontend {
        server hypedin-level-up_frontend:3000;
}

upstream level-up_backend {
        server hypedin-level-up_backend:3000;
}

upstream fateh-frontend {
	server hypedin-fateh_frontend:3000;
}

upstream fateh-backend {
	server hypedin-fateh_backend:3000;
}

server {
    listen 80;
    server_name www.bgmicampusmvp.com bgmicampusmvp.com;

    server_tokens off;

    location /.well-known/acme-challenge/ {
    	root /var/www/certbot;
    }

    location / {
        return 301 https://$host$request_uri;
    }
}

server {
    listen 443 ssl;
    server_name www.bgmicampusmvp.com bgmicampusmvp.com;

    server_tokens off;

    ssl_certificate /etc/letsencrypt/live/bgmicampusmvp.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/bgmicampusmvp.com/privkey.pem;

    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" always;
    add_header X-Frame-Options "DENY" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;

    # --- API Traffic ---
    location /api/ {
        client_max_body_size 100M;

        proxy_pass http://bgmi-backend;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # --- Socket.io Traffic (CRITICAL) ---
    location /socket.io/ {
        proxy_pass http://bgmi-backend;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection $connection_upgrade;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 86400s;
        proxy_send_timeout 86400s;
    }

    # --- Frontend Traffic ---
    location / {
        proxy_pass http://bgmi-frontend;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection $connection_upgrade;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
