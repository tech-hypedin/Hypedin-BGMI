name: Deploy to Hostinger VPS

on:
  push:
    branches:
      - main

env:
  BACKEND_IMAGE: hypedin-crm-backend:latest
  FRONTEND_IMAGE: hypedin-crm-frontend:latest

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 24

      - name: Type-check backend
        working-directory: ./backend
        run: |
          npm ci
          npm run build

      - name: Type-check frontend
        working-directory: ./frontend
        env:
          VITE_BACKEND_URL: ${{ secrets.VITE_BACKEND_URL }}
        run: |
          npm ci
          npm run build

      - name: Build backend image
        run: docker build -t $BACKEND_IMAGE ./backend

      - name: Build frontend image
        run: |
          docker build -t $FRONTEND_IMAGE \
            --build-arg VITE_BACKEND_URL=${{ secrets.VITE_BACKEND_URL }} \
            ./frontend

      - name: Save images to tarballs
        run: |
          docker save $BACKEND_IMAGE | gzip > backend-image.tar.gz
          docker save $FRONTEND_IMAGE | gzip > frontend-image.tar.gz

      - name: Upload images to VPS
        uses: appleboy/scp-action@v0.1.7
        with:
          host: ${{ secrets.SSH_HOST }}
          username: ${{ secrets.SSH_USER }}
          password: ${{ secrets.SSH_PASSWORD }}
          port: 22
          source: "backend-image.tar.gz,frontend-image.tar.gz"
          target: "~/Hypedin-CRM"

      - name: Deploy on VPS
        uses: appleboy/ssh-action@v1.0.3
        with:
          host: ${{ secrets.SSH_HOST }}
          username: ${{ secrets.SSH_USER }}
          password: ${{ secrets.SSH_PASSWORD }}
          port: 22
          script: |
            cd ~/Hypedin-CRM

            git fetch https://x-access-token:${{ secrets.GITHUB_TOKEN }}@github.com/Mitul82/Hypedin-CRM.git main
            git reset --hard FETCH_HEAD

            docker load < backend-image.tar.gz
            docker load < frontend-image.tar.gz
            rm backend-image.tar.gz frontend-image.tar.gz

            docker compose --env-file ./.env up -d
